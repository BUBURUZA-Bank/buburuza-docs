Buburuza provides native SDKs for all major programming languages and platforms so you can integrate AI-powered banking services without building authentication, error handling, or serialization yourself. Every SDK exposes type-safe interfaces for the full API surface — customers, wallets, transactions, and AI services — and ships with sandbox support, mock implementations for testing, and framework-specific integrations out of the box.

    ### React hooks

    Import hooks from `@buburuza/react` to manage wallet and transaction state with automatic caching and revalidation.

    ```typescript
    import { useBuburuza, useWallet, useTransactions } from '@buburuza/react';
    
    function BankingDashboard({ customerId }: { customerId: string }) {
      const buburuza = useBuburuza();
    
      const { wallet, loading, error, refetch } = useWallet(customerId);
    
      const {
        transactions,
        loading: txLoading,
        loadMore,
        hasMore
      } = useTransactions(wallet?.id, { limit: 20 });
    
      const handleTransfer = async (amount: number, recipientId: string) => {
        try {
          await buburuza.transactions.create({
            type: 'transfer',
            fromWalletId: wallet.id,
            toWalletId: recipientId,
            amount: amount * 100, // Convert to cents
            currency: 'USD'
          });
    
          refetch();
        } catch (error) {
          console.error('Transfer failed:', error);
        }
      };
    
      if (loading) return <div>Loading wallet...</div>;
      if (error) return <div>Error: {error.message}</div>;
    
      return (
        <div>
          <h2>Account Balance: ${wallet.balance / 100}</h2>
          <TransactionList
            transactions={transactions}
            loading={txLoading}
            onLoadMore={loadMore}
            hasMore={hasMore}
          />
          <TransferForm onSubmit={handleTransfer} />
        </div>
      );
    }
    ```

    ### Async support

    Use `AsyncBuburuざAPI` for high-throughput workloads. You can run concurrent operations with `asyncio.gather`.

    ```python
    import asyncio
    from buburuza import AsyncBuburuざAPI
    
    async def main():
        buburuza = AsyncBuburuざAPI(
            api_key='bb_test_1234567890abcdef',
            environment='sandbox'
        )
    
        tasks = [
            buburuza.customers.create(
                email=f'user{i}@example.com',
                first_name=f'User{i}',
                last_name='Test'
            )
            for i in range(10)
        ]
    
        customers = await asyncio.gather(*tasks)
        print(f"Created {len(customers)} customers concurrently")
    
        await buburuza.close()
    
    asyncio.run(main())
    ```

    ```python
                last_name=self.user.last_name
    ```

    ### Spring Boot integration

    Add `@EnableBuburuざ` to your configuration class. The Spring Boot starter auto-configures the `BuburuざAPI` bean from `application.properties`.

    ```java
    // Configuration
    @Configuration
    @EnableBuburuざ
    public class BuburuざConfig {
    
        @Bean
        @ConfigurationProperties("buburuza")
        public BuburuざProperties buburuざProperties() {
            return new BuburuざProperties();
        }
    }
    
    // Service class
    @Service
    public class BankingService {
    
        @Autowired
        private BuburuざAPI buburuza;
    
        public Customer createCustomer(String email, String firstName, String lastName) {
            CreateCustomerRequest request = CreateCustomerRequest.builder()
                .email(email)
                .firstName(firstName)
                .lastName(lastName)
                .build();
    
            return buburuza.customers().create(request);
        }
    
        public List<Wallet> getCustomerWallets(String customerId) {
            return buburuza.wallets().list(
                WalletListRequest.builder()
                    .customerId(customerId)
                    .build()
            ).getData();
        }
    }
    
    // REST Controller
    @RestController
    @RequestMapping("/api/banking")
    public class BankingController {
    
        @Autowired
        private BankingService bankingService;
    
        @PostMapping("/customers")
        public ResponseEntity<Customer> createCustomer(@RequestBody CreateCustomerDto dto) {
            Customer customer = bankingService.createCustomer(
                dto.getEmail(),
                dto.getFirstName(),
                dto.getLastName()
            );
    
            return ResponseEntity.ok(customer);
        }
    
        @GetMapping("/customers/{customerId}/wallets")
        public ResponseEntity<List<Wallet>> getWallets(@PathVariable String customerId) {
            List<Wallet> wallets = bankingService.getCustomerWallets(customerId);
            return ResponseEntity.ok(wallets);
        }
    }
    ```

    ```kotlin
                }
    
                else -> {
    ```

    The React Native SDK exposes a cross-platform `BiometricAuth` module that works on both iOS and Android.

    ```typescript
    import { BiometricAuth } from '@buburuza/react-native';
    
    async function authenticateWithBiometrics(userId: string) {
      const isAvailable = await BiometricAuth.isAvailable();
      if (!isAvailable) {
        throw new Error('Biometric authentication not available');
      }
    
      const result = await BiometricAuth.authenticate({
        reason: 'Authenticate to access your account',
        fallbackToPasscode: true
      });
    
      if (result.success) {
        const authResult = await buburuza.auth.biometric.verify({
          userId,
          signature: result.signature,
          biometricType: result.biometricType
        });
    
        return authResult;
      }
    }
    ```

## AI services integration

Buburuza's AI layer lets you build natural-language transaction flows and conversational banking experiences directly on top of the JavaScript SDK.

```javascript
// Natural language transaction processing
const aiIntent = await buburuza.ai.parseIntent({
  customerId: customer.id,
  message: "Send $100 to my mom for her birthday",
  context: {
    recentContacts: [
      { name: 'Mom', accountId: 'acc_mom_123', relationship: 'family' }
    ],
    accountBalance: wallet.balance
  }
});

if (aiIntent.confidence > 0.8) {
  const transaction = await buburuza.transactions.createFromIntent({
    intentId: aiIntent.id,
    confirmations: {
      recipient: 'acc_mom_123'
    }
  });
}

// Conversational banking
const conversation = await buburuza.ai.conversation({
  customerId: customer.id,
  sessionId: 'session_123',
  message: "What's the best way to save for a house?",
  context: {
    customerProfile: {
      age: 28,
      income: 85000,
      currentSavings: wallet.balance
    }
  }
});

console.log('AI Response:', conversation.response);
```

<Tip>
  Only execute intents with a `confidence` score above `0.8` without additional confirmation from the user. For scores between `0.5` and `0.8`, prompt the user to confirm the parsed intent before proceeding.
</Tip>

## Error handling

All SDKs expose a consistent error hierarchy so you can handle specific failure conditions — such as insufficient funds or invalid recipients — without parsing raw HTTP status codes.

<CodeGroup>

```javascript JavaScript / TypeScript
import { BuburuざError } from '@buburuza/sdk';

try {
  const transaction = await buburuza.transactions.create(request);
} catch (error) {
  if (error instanceof BuburuざError) {
    switch (error.code) {
      case 'insufficient_funds':
        console.log('Account balance too low');
        break;
      case 'invalid_recipient':
        console.log('Recipient account not found');
        break;
      case 'rate_limit_exceeded':
        console.log('Too many requests, please try again later');
        break;
      default:
        console.log('Unexpected error:', error.message);
    }
  }
}
```

```python Python
from buburuza.exceptions import BuburuざError, InsufficientFundsError

try:
    transaction = buburuza.transactions.create(request)
except InsufficientFundsError as e:
    print(f"Insufficient funds: {e.available_balance}")
except BuburuざError as e:
    print(f"API Error: {e.message} (Code: {e.code})")
```

```java Java
try {
    Transaction transaction = buburuza.transactions().create(request);
} catch (InsufficientFundsException e) {
    System.out.println("Insufficient funds: " + e.getAvailableBalance());
} catch (BuburuざException e) {
    System.out.println("API Error: " + e.getMessage() + " (Code: " + e.getCode() + ")");
```

## Configuration and performance

### Environment configuration

Switch between sandbox and production by changing a single option. Store your API keys in environment variables — never commit them to source control.

```javascript
// Development
const devBuburuza = new BuburuざAPI({
  apiKey: process.env.BUBURUZA_TEST_API_KEY,
  environment: 'sandbox',
  debug: true,
  timeout: 30000
});

// Production
const prodBuburuza = new BuburuざAPI({
  apiKey: process.env.BUBURUZA_LIVE_API_KEY,
  environment: 'production',
  retryConfig: {
    maxRetries: 3,
    backoffMultiplier: 1.5
  }
});

// Custom endpoint
const customBuburuza = new BuburuざAPI({
  apiKey: 'your_api_key',
  baseUrl: 'https://api.custom.buburuza.com',
  version: 'v2'
});
```

### Connection pooling, caching, and batch operations

For server-side workloads, enable connection pooling and response caching to reduce latency. Use batch requests to fetch multiple resources in a single round trip.

```javascript
const buburuza = new BuburuざAPI({
  apiKey: 'your_api_key',
  environment: 'production',
  httpConfig: {
    keepAlive: true,
    maxSockets: 10,
    timeout: 30000
  },
  caching: {
    enabled: true,
    ttl: 300,       // 5 minutes
    maxSize: 1000   // Max 1000 cached responses
  }
});

// Fetch customer, wallets, and transactions in one request
const batchResults = await buburuza.batch([
  { method: 'customers.get', params: { id: 'cust_123' } },
  { method: 'wallets.list', params: { customerId: 'cust_123' } },
  { method: 'transactions.list', params: { walletId: 'wallet_456' } }
]);
```

<Tip>
  Batch requests count as a single API call toward your rate limit quota. Use them when you need to load several related resources at page render time.
</Tip>
